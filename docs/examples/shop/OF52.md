# (OF52) Export offers CSV or JSON file asynchronously

```php
use Mirakl\MMP\Shop\Client\ShopApiClient;
use Mirakl\MMP\Shop\Request\Offer\Async\Export\OffersExportAsyncRequest;

$api = new ShopApiClient('https://your-marketplace.mirakl.net/api', 'your-api-key', 'your-shop-id');

$request = new OffersExportAsyncRequest();
$request->setExportType('text/csv')
    ->setChannelCodes(['web', 'mobile'])
    ->setLastRequestDate(new \DateTime('2024-01-01T00:00:00Z'))
    ->setIncludeInactiveOffers(false);

$result = $api->createOffersExportAsync($request);
// $result => @see \Mirakl\MMP\Common\Domain\Offer\Async\Export\OffersExportAsyncTrackingResult

echo $result->getTrackingId();
```
