# (P11) List offers for each product given

```php
use Mirakl\MMP\Shop\Client\ShopApiClient;
use Mirakl\MMP\Shop\Request\Product\Offer\GetOffersOnProductsRequest;

$api = new ShopApiClient('https://your-marketplace.mirakl.net/api', 'your-api-key', 'your-shop-id');

$request = new GetOffersOnProductsRequest(['product-sku-001']);
$request->setChannelCodes(['CHANNEL_WEB'])
    ->setOfferStateCodes(['11'])
    ->setAllOffers(true)
    ->setMax(10);

$result = $api->getOffersOnProducts($request);
// $result => @see \Mirakl\MMP\Shop\Domain\Collection\Product\Offer\ProductWithOffersCollection

foreach ($result as $product) {
    echo $product->getProduct()->getSku();
}
```
