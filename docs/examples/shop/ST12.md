# (ST12) List items to ship

```php
use Mirakl\MMP\Shop\Client\ShopApiClient;
use Mirakl\MMP\Shop\Request\Shipment\GetItemsToShipRequest;

$api = new ShopApiClient('https://your-marketplace.mirakl.net/api', 'your-api-key', 'your-shop-id');

$request = new GetItemsToShipRequest();
$request->addOrderId('order-12345')
    ->setShippingDateFrom(new \DateTime('2024-01-01T00:00:00Z'))
    ->setShippingDateTo(new \DateTime('2024-01-31T23:59:59Z'));

$result = $api->getItemsToShip($request);
// $result => @see \Mirakl\MMP\Common\Domain\Collection\SeekableCollection

foreach ($result->getCollection() as $item) {
    echo $item->getOrderId() . ' — ' . $item->getShippingTypeCode();
}

// Fetch next page
if ($nextToken = $result->getNextPageToken()) {
    $request = new GetItemsToShipRequest();
    $request->setPageToken($nextToken);
    $result = $api->getItemsToShip($request);
}
```
