# (H11) List Catalog categories

```php
use Mirakl\MCI\Shop\Client\ShopApiClient;
use Mirakl\MCI\Shop\Request\Hierarchy\GetHierarchiesRequest;

$api = new ShopApiClient('https://your-marketplace.mirakl.net/api', 'your-api-key', 'your-shop-id');

$request = new GetHierarchiesRequest();
$request->setHierarchyCode('ROOT_CATEGORY')
    ->setMaxLevel(2);

$result = $api->getHierarchies($request);
// $result => @see \Mirakl\MCI\Common\Domain\Collection\HierarchyCollection

foreach ($result as $hierarchy) {
    echo $hierarchy->getCode() . ' — ' . $hierarchy->getLabel();
}
```
