# (PM11) Get the product attribute configuration

```php
use Mirakl\MCI\Shop\Client\ShopApiClient;
use Mirakl\MCI\Shop\Request\Attribute\GetAttributesRequest;

$api = new ShopApiClient('https://your-marketplace.mirakl.net/api', 'your-api-key', 'your-shop-id');

$request = new GetAttributesRequest();
$request->setHierarchyCode('electronics')
    ->setMaxLevel(3)
    ->setWithRoles(true)
    ->setAllOperatorAttributes(true);

$result = $api->getAttributes($request);
// $result => @see \Mirakl\MCI\Common\Domain\Collection\AttributeCollection

foreach ($result as $attribute) {
    echo $attribute->getCode() . ' — ' . $attribute->getLabel();
}
```
