# (PR04) Update a promotion

```php
use Mirakl\MMP\Common\Domain\Collection\Promotion\UpdatePromotion\PublicDescriptionCollection;
use Mirakl\MMP\Common\Domain\Promotion\UpdatePromotion\PublicDescription;
use Mirakl\MMP\Shop\Client\ShopApiClient;
use Mirakl\MMP\Shop\Request\Promotion\UpdatePromotionRequest;

$api = new ShopApiClient('API_URL', 'API_KEY', 'SHOP_ID');

$description = (new PublicDescription())
    ->setLocale('en_US')
    ->setValue('Buy 2, get 1 free — now extended to all categories');

$request = new UpdatePromotionRequest('PROMO-BUY2GET1');
$request->setEndDate(new \DateTime('2024-09-30T23:59:59Z'))
    ->setChannelCodes(['WEB'])
    ->setPublicDescriptions(new PublicDescriptionCollection([$description]));

$api->updatePromotion($request);
// No return value
```
