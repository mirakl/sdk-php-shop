# (OR11) List orders with pagination

```php
use Mirakl\MMP\Shop\Client\ShopApiClient;
use Mirakl\MMP\Shop\Request\Order\Get\GetOrdersRequest;

$api = new ShopApiClient('https://your-marketplace.mirakl.net/api', 'your-api-key', 'your-shop-id');

$request = new GetOrdersRequest();
$request->setOrderStates(['SHIPPING', 'SHIPPED'])
    ->setOrderIds(['order-12345'])
    ->setStartUpdateDate(new \DateTime('2024-01-01T00:00:00Z'));

$result = $api->getOrders($request);
// $result => @see \Mirakl\MMP\Shop\Domain\Collection\Order\ShopOrderCollection

foreach ($result as $order) {
    echo $order->getId() . ' — ' . $order->getStatus()->getState();
}
echo 'Total: ' . $result->getTotalCount();
```
