# (TL02) List transaction lines

```php
use Mirakl\MMP\Shop\Client\ShopApiClient;
use Mirakl\MMP\Shop\Request\Payment\Transaction\TransactionLineRequest;

$api = new ShopApiClient('https://your-marketplace.mirakl.net/api', 'your-api-key', 'your-shop-id');

$request = new TransactionLineRequest();
$request->setTransactionType(['ORDER_AMOUNT', 'COMMISSION_FEE'])
    ->setPaymentState(['PENDING'])
    ->setDateCreatedFrom(new \DateTime('-30 days'));

$result = $api->getTransactionLine($request);
// $result => @see \Mirakl\MMP\Common\Domain\Collection\SeekableCollection

foreach ($result->getCollection() as $transaction) {
    echo $transaction->getId() . ': ' . $transaction->getAmount() . ' ' . $transaction->getCurrencyIsoCode();
}

// Fetch the next page using the page token
if ($result->getNextPageToken()) {
    $request = new TransactionLineRequest();
    $request->setPageToken($result->getNextPageToken());
    $result = $api->getTransactionLine($request);
}
```
