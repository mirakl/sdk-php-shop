# (ST11) List shipments

```php
use Mirakl\MMP\Shop\Client\ShopApiClient;
use Mirakl\MMP\Common\Domain\Shipment\ShipmentStatus;
use Mirakl\MMP\Shop\Request\Shipment\GetShipmentsRequest;

$api = new ShopApiClient('https://your-marketplace.mirakl.net/api', 'your-api-key', 'your-shop-id');

$request = new GetShipmentsRequest();
$request->addShipmentStateCode(ShipmentStatus::SHIPPED)
    ->addOrderId('order-12345')
    ->setLastUpdatedFrom(new \DateTime('2024-01-01T00:00:00Z'));

$result = $api->getShipments($request);
// $result => @see \Mirakl\MMP\Common\Domain\Collection\SeekableCollection

foreach ($result->getCollection() as $shipment) {
    echo $shipment->getId() . ' — ' . $shipment->getStatus();
}

// Fetch next page
if ($nextToken = $result->getNextPageToken()) {
    $request = new GetShipmentsRequest();
    $request->setPageToken($nextToken);
    $result = $api->getShipments($request);
}
```
