# (PR03) Create a promotion

```php
use Mirakl\MMP\Common\Domain\Collection\Promotion\CreatePromotions\PublicDescriptionCollection;
use Mirakl\MMP\Common\Domain\Promotion\CreatePromotions\PublicDescription;
use Mirakl\MMP\Common\Domain\Promotion\CreatePromotions\Reward\PercentageOffReward;
use Mirakl\MMP\Common\Domain\Promotion\CreatePromotions\Trigger\MinimumQuantityTrigger;
use Mirakl\MMP\Shop\Client\ShopApiClient;
use Mirakl\MMP\Shop\Request\Promotion\CreatePromotionsRequest;

$api = new ShopApiClient('https://your-marketplace.mirakl.net/api', 'your-api-key', 'your-shop-id');

$reward = (new PercentageOffReward())
    ->setPercentageOff(100.0)
    ->setMaxQuantityToDiscount(1);

$trigger = (new MinimumQuantityTrigger())
    ->setQuantityPurchased(2);

$description = (new PublicDescription())
    ->setLocale('en_US')
    ->setValue('Buy 2, get 1 free');

$request = new CreatePromotionsRequest();
$request->setInternalId('PROMO-BUY2GET1')
    ->setStartDate(new \DateTime('2024-06-01T00:00:00Z'))
    ->setEndDate(new \DateTime('2024-08-31T23:59:59Z'))
    ->setPublicDescriptions(new PublicDescriptionCollection([$description]))
    ->setReward($reward)
    ->setTrigger($trigger);

$api->createPromotions($request);
// No return value
```
