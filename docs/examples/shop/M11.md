# (M11) List all threads

```php
use Mirakl\MMP\Shop\Client\ShopApiClient;
use Mirakl\MMP\OperatorShop\Request\Message\GetThreadsRequest;

$api = new ShopApiClient('https://your-marketplace.mirakl.net/api', 'your-api-key', 'your-shop-id');

$request = new GetThreadsRequest();
$request->setEntityType('MMP_ORDER')
    ->setUpdatedSince(new \DateTime('2024-01-15T10:30:00Z'))
    ->setWithMessages(true);

$result = $api->getThreads($request);
// $result => @see \Mirakl\MMP\Common\Domain\Collection\SeekableCollection

foreach ($result->getCollection() as $thread) {
    echo $thread->getId() . ' — ' . $thread->getTopic()->getValue();
}

$nextPageToken = $result->getNextPageToken();
```
