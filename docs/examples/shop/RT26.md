# (RT26) Mark a return as compliant or non compliant

```php
use Mirakl\MMP\Common\Domain\Collection\Returns\Workflow\MarkReturnComplianceCollection;
use Mirakl\MMP\Common\Domain\Returns\Workflow\MarkReturnCompliance;
use Mirakl\MMP\Common\Domain\Returns\Workflow\MarkReturnLineCompliance;
use Mirakl\MMP\Shop\Client\ShopApiClient;
use Mirakl\MMP\Shop\Request\Returns\MarkReturnsComplianceRequest;

$api = new ShopApiClient('https://your-marketplace.mirakl.net/api', 'your-api-key', 'your-shop-id');

$line = new MarkReturnLineCompliance();
$line->setOrderLineId('ORDER-001-A-1')
    ->setCompliant(false)
    ->setNonComplianceReasonCode('RETURN_NON_COMPLIANT_OTHER')
    ->setNonComplianceAdditionalInfo('Item damaged on arrival');

$return = new MarkReturnCompliance();
$return->setId('9d6290c3-8b25-43c0-b448-914d0aab38c0')
    ->setReturnLines([$line]);

$request = new MarkReturnsComplianceRequest(new MarkReturnComplianceCollection([$return]));

$result = $api->markReturnsCompliance($request);
// $result is a ReturnWorkflowResponse
// $result->getReturnSuccess() / $result->getReturnErrors()
```
