# (RT01) Create returns

```php
use Mirakl\MMP\Shop\Client\ShopApiClient;
use Mirakl\MMP\Shop\Domain\Collection\Returns\ReturnCreationCollection;
use Mirakl\MMP\Shop\Domain\Collection\Returns\ReturnCreationReturnLineCollection;
use Mirakl\MMP\Shop\Domain\Returns\ReturnCreation;
use Mirakl\MMP\Shop\Domain\Returns\ReturnCreationReturnLine;
use Mirakl\MMP\Shop\Request\Returns\CreateReturnsRequest;

$api = new ShopApiClient('https://your-marketplace.mirakl.net/api', 'your-api-key', 'your-shop-id');

$returnLine = new ReturnCreationReturnLine();
$returnLine->setOrderLineId('ORDER-001-A-1')
    ->setQuantity(2)
    ->setReasonCode('RETURN_CHANGED_MIND');

$returnLines = new ReturnCreationReturnLineCollection();
$returnLines->add($returnLine);

$return = new ReturnCreation();
$return->setOrderCommercialId('ORDER-001')
    ->setReasonCode('RETURN_CHANGED_MIND')
    ->setMethodCode('RETURN_METHOD_BY_MAIL')
    ->setReturnLines($returnLines);

$returns = new ReturnCreationCollection();
$returns->add($return);

$request = new CreateReturnsRequest($returns);

$result = $api->createReturns($request);
// $result is a ReturnCreationResponse
// $result->getReturnSuccess() → MiraklReturnCollection (Mirakl\MMP\Common\Domain\Returns\MiraklReturn)
// $result->getReturnErrors()  → ReturnCreationErrorCollection
```
